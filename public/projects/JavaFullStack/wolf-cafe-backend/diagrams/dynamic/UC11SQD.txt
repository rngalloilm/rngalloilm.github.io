title: Use Case - Create an Account

User->AuthController: register(RegisterDto)
AuthController->AuthService: register(RegisterDto)
AuthService->UserRepository: existsByUsername(username)
UserRepository-->AuthService: Boolean (exists)

alt Username exists
    AuthService-->AuthController: "Username already exists"
    AuthController-->User: "Username already exists"
else Username available
    AuthService->UserRepository: existsByEmail(email)
    UserRepository-->AuthService: Boolean (exists)
    
    alt Email exists
        AuthService-->AuthController: "Email already exists"
        AuthController-->User: "Email already exists"
    else Email available
        alt Password valid
            AuthService->UserRepository: save(user)
            UserRepository-->AuthService: User
            AuthService-->AuthController: "User created successfully"
            AuthController-->User: "Account created and logged in"
        else Password invalid
            AuthService-->AuthController: "Invalid password"
            AuthController-->User: "Invalid password, try again"
        end
    end
end
